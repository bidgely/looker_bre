view: itron_enrollment_files {
  derived_table: {
    sql: with meter_events as (
          select event_id, uuid, gateway_ordinal, meter_ordinal, start_timestamp, end_timestamp, event_timestamp, token,
          SUBSTRING(token FROM 7 FOR GREATEST(POSITION('_AMI' IN token) - 7, 0)) AS meter_id,
          row_number() over (partition by uuid, meter_id order by event_timestamp) as rid_start,
          row_number() over (partition by uuid, meter_id order by event_timestamp desc) as rid_end,
          case when end_timestamp > 0 then true else false end as closed_meter_event_available
          from reporting.enrollment_events
          where event_type like 'METER%' and meter_ordinal > 0
          and uuid in (select uuid from user_meta_data where pilot_id in ('10112', '10014', '10117', '10118', '10126', '10223', '10224'))
          --and uuid = 'e0341b49-653c-4d44-b5e3-c19f4a816658'
          ),

        meter_start_events as (
          select *
          from meter_events
          where rid_start = 1),

        meter_end_events as (
          select *
          from meter_events
          where rid_end = 1 and closed_meter_event_available = true)

        select a.uuid, c.pilot_id, a.meter_id, a.start_timestamp, b.end_timestamp,
        case when b.end_timestamp is null and a.start_timestamp <= EXTRACT(EPOCH FROM GETDATE()) then true
             when b.end_timestamp > EXTRACT(EPOCH FROM GETDATE()) then true else false end as is_active_meter
        from meter_start_events a
        left join meter_end_events b
        on a.meter_id = b.meter_id
        left join public.user_meta_data c on a.uuid = c.uuid
        where {%condition active_meter%} is_active_meter {%endcondition%}
      ;;
  }

  filter: active_meter {
    type: yesno
  }

  dimension: uuid {
    type: string
    sql: ${TABLE}.uuid ;;
  }

  dimension: pilot_id {
    type: string
    sql: ${TABLE}.pilot_id ;;
  }

  dimension: pilot_name {
    type: string
    sql: case when ${pilot_id} =  '10112' then 'TECO'
              when ${pilot_id} =  '10014' then 'SMUD'
              when ${pilot_id} =  '10224' then 'CNP'. --updated from 10015 UAT
              when ${pilot_id} =  '10117' then 'SCE'
              when ${pilot_id} =  '10118' then 'Pecan ST'
              when ${pilot_id} =  '10126' then 'Hydro One'
              when ${pilot_id} =  '10223' then 'LUMA'
              else 'Others' end
        ;;
  }

  dimension: start_time_epoch {
    description: "epoch of the start time of meter"
    type: number
    sql: ${TABLE}.start_timestamp ;;
  }

  dimension: end_time_epoch {
    description: "epoch of the end timen of meter"
    type: number
    sql: ${TABLE}.end_timestamp ;;
  }

  #will update timezone values once itron gets updated in pilot table

  dimension_group: start_time {
    type: time
    timeframes: [raw, time, date, week, month]
    sql: CONVERT_TIMEZONE('UTC', 'US/Eastern', DATEADD(SECOND, ${start_time_epoch}, '1970-01-01')) ;;
    #sql: convert_timezone('UTC',${pilot_consolidated_product_types.time_zone}, dateadd( SECOND, ${start_time_epoch}, '1970-1-1' )) ;;
  }

  dimension_group: end_time {
    type: time
    timeframes: [raw, time, date, week, month]
    sql: convert_timezone('UTC','US/Eastern', dateadd( SECOND, ${end_time_epoch}, '1970-1-1' )) ;;
    #sql: convert_timezone('UTC',${pilot_consolidated_product_types.time_zone}, dateadd( SECOND, ${end_time_epoch}, '1970-1-1' )) ;;
  }

  dimension: meter_id {
    description: "meter id for the user"
    type: string
    primary_key: yes
    sql: ${TABLE}.meter_id ;;
  }

  dimension: is_active_meter {
    description: "current status for the meter"
    type: yesno
    sql: ${TABLE}.is_active_meter ;;
  }

  measure: count_users {
    type: count_distinct
    sql:  ${uuid};;
    drill_fields: [uuid]
  }

  measure: count_active_users {
    type: count_distinct
    sql:  ${uuid};;
    drill_fields: [uuid]
    filters: {
      field: is_active_meter
      value: "Yes"
    }
  }

  measure: count_meters {
    type: count_distinct
    sql:  ${meter_id};;
    drill_fields: [meter_id]
  }

  measure: count_active_meters {
    type: count_distinct
    sql:  ${meter_id};;
    drill_fields: [meter_id]
    filters: {
      field: is_active_meter
      value: "Yes"
    }
  }

  measure: count_enrolled_meters {
    type: count_distinct
    sql: CASE
          WHEN ${start_time_date} = ${date_series.continous_date}
          THEN ${TABLE}.meter_id
       END ;;
  }

  measure: count_unenrolled_meters {
    type: count_distinct
    sql: CASE
          WHEN ${end_time_date} = ${date_series.continous_date}
          THEN ${TABLE}.meter_id
       END ;;
  }

}

view: itron_config_issued {
  derived_table: {
    sql:  with base as (
            select
              uuid,
              start_date_for_config_generation,
              end_date_for_config_generation,
              status,
              min(last_updated_timestamp) as first_update_timestamp,
              max(last_updated_timestamp) as latest_update_timestamp,
              count(distinct last_updated_timestamp) as total_entries,
              count(distinct case when retry_count > 0 then last_updated_timestamp end) as total_entries_with_retries,
              sum(retry_count) as sum_of_retry_counts
            from meter_config_update_request
            --where uuid in ('01146133-d43b-4e7c-a949-5048a27ed363', 'bfad5e02-1e11-4be1-9d75-f273304300b9')
            group by 1, 2, 3, 4
          )
          select
            base.*,

            -- column_a: REQUEST_SENT timestamp per config
            min(case when status = 'REQUEST_SENT'
                then first_update_timestamp end)
                over(partition by uuid, start_date_for_config_generation, end_date_for_config_generation) as request_sent_ts,

            -- column_b: COMPLETED/COMPLETED_WITH_ERRORS timestamp per config
            min(case when status in ('COMPLETED', 'COMPLETED_WITH_ERRORS')
                then first_update_timestamp end)
                over(partition by uuid, start_date_for_config_generation, end_date_for_config_generation) as final_ts

          from base
          order by uuid, start_date_for_config_generation, status
          ;;
}

  dimension: uuid {
    type: string
    sql: ${TABLE}.uuid ;;
  }

  dimension: status {
    type: string
    sql: ${TABLE}.status ;;
  }

  dimension: start_date_for_config_generation {
    type: number
    sql: ${TABLE}.start_date_for_config_generation ;;
  }

  dimension: end_date_for_config_generation {
    type: number
    sql: ${TABLE}.end_date_for_config_generation ;;
  }

  dimension: config_status_key {
    primary_key: yes
    hidden: yes
    type: string
    sql: ${uuid} || '-' || ${start_date_for_config_generation} || '-' || ${end_date_for_config_generation} || '-' || ${status} ;;
  }

  dimension: config_key {
    hidden: yes
    type: string
    sql: MD5(${uuid} || '-' || ${start_date_for_config_generation} || '-' || ${end_date_for_config_generation}) ;;
  }

# Convert epoch (10-digit seconds) to readable date
  dimension_group: start_date {
    type: time
    timeframes: [raw, date, week, month, year]
    datatype: epoch
    sql: ${TABLE}.start_date_for_config_generation ;;
  }

  dimension_group: end_date {
    type: time
    timeframes: [raw, date, week, month, year]
    datatype: epoch
    sql: ${TABLE}.end_date_for_config_generation ;;
  }


  dimension: first_update_timestamp {
    type: number
    sql: ${TABLE}.first_update_timestamp ;;
  }

  dimension: latest_update_timestamp {
    type: number
    sql: ${TABLE}.latest_update_timestamp ;;
  }

# Convert epoch (13-digit seconds) to readable date
  dimension_group: first_update {
    type: time
    timeframes: [raw, date, week, month, year]
    datatype: epoch
    sql: ${TABLE}.first_update_timestamp / 1000 ;;
  }

  dimension_group: latest_update {
    type: time
    timeframes: [raw, date, week, month, year]
    datatype: epoch
    sql: ${TABLE}.latest_update_timestamp  / 1000 ;;
  }

  dimension: total_entries {
    type: number
    sql: ${TABLE}.total_entries ;;
  }

  dimension: total_entries_with_retries {
    type: number
    sql: ${TABLE}.total_entries_with_retries ;;
  }

  dimension: sum_of_retry_counts {
    type: number
    sql: ${TABLE}.sum_of_retry_counts ;;
  }

  dimension: uuid_timestamp_key {
    #primary_key: yes
    hidden: yes
    type: string
    sql: ${uuid} || '-' || ${latest_update_timestamp} ;;
  }


  dimension: request_sent_ts {
    type: number
    sql: ${TABLE}.request_sent_ts ;;
  }

  dimension: final_ts {
    type: number
    sql: ${TABLE}.final_ts ;;
  }

  dimension: time_required_seconds {
    type: number
    sql: CASE
         WHEN ${status} = 'REQUEST_SENT'
           AND ${request_sent_ts} IS NOT NULL
           AND ${final_ts} IS NOT NULL
         THEN (${final_ts} - ${request_sent_ts}) / 1000
       END ;;
  }

  measure: average_time_required_seconds {
    type: number
    sql: AVG(CASE WHEN ${status} = 'REQUEST_SENT'
            AND ${request_sent_ts} IS NOT NULL
            AND ${final_ts} IS NOT NULL
       THEN (${final_ts} - ${request_sent_ts}) / 1000 END) ;;
    value_format_name: decimal_2
    label: "Avg Time Required (seconds)"
  }

  measure: average_time_required_min {
    type: number
    sql: ${average_time_required_seconds} / 60.0 ;;
    value_format_name: decimal_2
    label: "Avg Time Required (minutes)"
  }

  measure: average_time_required_hh_mm_ss {
    type: string
    sql:
    TO_CHAR(
      DATEADD(
        second,
        ${average_time_required_seconds},
        TIMESTAMP '1970-01-01'
      ),
      'HH24:MI:SS'
    ) ;;
    label: "Avg Time Required (HH:MM:SS)"
  }

  measure: total_entries_sum {
    type: sum
    sql:  ${total_entries} ;;
  }

  measure: total_entries_with_retries_sum {
    type: sum
    sql:  ${total_entries_with_retries} ;;
  }

  measure: sum_retry_counts {
    type: sum
    sql:  ${sum_of_retry_counts} ;;
  }

  measure: count_users_config_issued {
    type: count_distinct
    sql: CASE WHEN ${first_update_date} = ${date_series.continous_date} THEN ${uuid_timestamp_key} END ;;
  }

  measure: count_users_per_day {
    type: count_distinct
    sql: CASE WHEN ${first_update_date} = ${date_series.continous_date} THEN ${uuid} END ;;
  }

}

view: itron_meter_config_update_request {
  derived_table: {
    sql:
      WITH last_event AS (
        SELECT
          request_id,
          status,
          ROW_NUMBER() OVER (PARTITION BY request_id ORDER BY last_updated_timestamp DESC) AS rn
        FROM public.meter_config_update_request
      )
      SELECT
        m.request_id,
        MAX(m.uuid) AS uuid,
        MAX(m.esn_list) AS meter_esn,
        MAX(m.pilot_id) AS pilot_id,
        MAX(m.partner_id) AS partner_id,
        MAX(m.feature) AS feature,
        MAX(m.hardware_version) AS hardware_version,
        MAX(m.country_code) AS country_code,
        MAX(m.retry_count) AS retry_count,
        MAX(le.status) AS final_status,
        MAX(CASE WHEN m.status = 'CONFIG_GENERATED' THEN 1 ELSE 0 END) AS has_config_generated,
        COALESCE(
          MIN(CASE WHEN m.status = 'CONFIG_GENERATED' THEN m.last_updated_timestamp END),
          MIN(m.last_updated_timestamp)
        ) AS issued_at_ms,
        MAX(CASE WHEN m.status IN ('INITIALIZING','PREPARING','REQUEST_SENT','COMPLETED','FAILED')
                 THEN 1 ELSE 0 END) AS is_confirmed_sent,
        MIN(CASE WHEN m.status IN ('INITIALIZING','PREPARING','REQUEST_SENT','COMPLETED','FAILED')
                 THEN m.last_updated_timestamp END) AS confirmed_at_ms,
        MAX(CASE WHEN m.status = 'FAILED' THEN 1 ELSE 0 END) AS is_failed,
        MAX(CASE
              WHEN m.status = 'FAILED' AND m.remarks LIKE 'Patch request time out%' THEN 'api_timeout'
              WHEN m.status = 'FAILED' AND m.remarks LIKE 'Config confirmation message not received%' THEN 'no_confirmation_received'
              WHEN m.status = 'FAILED' THEN 'other'
            END) AS failure_reason,
        MAX(CASE WHEN m.status = 'COMPLETED' THEN 1 ELSE 0 END) AS is_completed,
        MIN(CASE WHEN m.status = 'COMPLETED' THEN m.last_updated_timestamp END) AS completed_at_ms
      FROM public.meter_config_update_request m
      JOIN last_event le ON le.request_id = m.request_id AND le.rn = 1
      GROUP BY m.request_id
     ;;
  }

  dimension: request_id {
    primary_key: yes
    type: string
    sql: ${TABLE}.request_id ;;
  }

  dimension: uuid {
    type: string
    sql: ${TABLE}.uuid ;;
  }

  dimension: meter_esn {
    label: "Meter ESN"

  dimension: pilot_id {
    type: number
    sql: ${TABLE}.pilot_id ;;

  dimension: partner_id {
    type: string
    sql: ${TABLE}.partner_id ;;
  }

  dimension: feature {
    type: string
    sql: ${TABLE}.feature ;;

  dimension: hardware_version {
    type: string

  dimension: country_code {
    type: string

  dimension: retry_count {
    label: "Retry Count"
    description: "Attempt number for this meter's configuration process; constant per request_id. A handful of meters climb into the high teens/thirties — a persistent fail/retry loop worth monitoring."
    type: number
    sql: ${TABLE}.retry_count ;;
  }

  dimension: is_retry {
    label: "Is Retry"
    description: "True if this request is itself a retry attempt (retry_count > 0)"
    type: yesno
    sql: ${TABLE}.retry_count > 0 ;;
  }

  dimension: final_status {
    label: "Final Status"
    description: "The last status this request reached, by most recent event timestamp — e.g. COMPLETED, FAILED, REQUEST_SENT, CONFIG_GENERATED, PENDING"
    type: string
    sql: ${TABLE}.final_status ;;
  }

  dimension: has_config_generated {

  dimension: is_confirmed_sent {

  measure: average_confirmation_time_hours {

  }

  measure: confirmation_rate {
    label: "Confirmation Rate"
    description: "Share of issued requests that were eventually confirmed sent"

    value_format_name: percent_1
  }

  measure: count_retries {

  }

  measure: retry_rate {

    sql: 1.0 * ${count_retries} / NULLIF(${count_configuration_requests_issued}, 0) ;;

  measure: count_failures {

    type: average
    sql: ${successful_configuration_duration_hours} ;;
    value_format_name: decimal_1
  }
